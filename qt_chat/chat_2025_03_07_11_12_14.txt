### 步骤 1：继承 QMainWindow 并重写 paintEvent 方法
通过继承 QMainWindow 并在其中添加自定义绘制，可以在主窗口上绘制一个圆角矩形。

```cpp
#include <QMainWindow>
#include <QPainter>

class CustomMainWindow : public QMainWindow {
    Q_OBJECT

public:
    explicit CustomMainWindow(QWidget *parent = nullptr) : QMainWindow(parent) {
        // 设置窗口的标题栏样式（如果需要）
        setWindowTitle("自定义标题");
        // 隐藏默认标题栏，如果不需要的话
        setWindowFlag(Qt::FramelessWindowHint);
        // 创建并设置自定义标题栏
        QWidget *titleBar = new QWidget(this);
        titleBar->setObjectName("titleBar");
        this->setMenuBar(titleBar);  // 假设 titleBar 是菜单栏
    }

protected:
    void paintEvent(QPaintEvent *event) {
        QMainWindow::paintEvent(event);

        QPainter painter(viewport());
        painter.setRenderHint(QPainter::Antialiasing);
        QColor background = palette().window().color();

        // 计算窗口的大小，减去可能被标题栏或其他部件占用的部分
        int border = 10; // 圆角半径
        QRect rect(0, titleBar->height(), width(), height() - titleBar->height());

        painter.setPen(Qt::NoPen);
        painter.setBrush(QBrush(background));
        painter.drawRoundedRect(rect, border, border); // 绘制圆角矩形背景

        // 重绘其他部件，确保标题栏和其他内容显示在背景之上
        Q_UNUSED(event);
    }

private:
    QWidget *titleBar = nullptr;
};
```
